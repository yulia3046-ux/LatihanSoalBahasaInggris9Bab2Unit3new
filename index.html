Konfigurasi Google Apps Script (GAS)

Petunjuk Guru / Penguji:
  1. Buat Spreadsheet baru di Google Sheets.
  2. Klik menu Extensions (Ekstensi) > Apps Script.
  3. Hapus seluruh kode yang ada, lalu **salin dan tempel kode GS** di bawah ini.
  4. Klik **Deploy (Terapkan) > New deployment (Terapkan baru)**.
  5. Pilih type: **Web App**, set *Who has access*: **Anyone (Siapa saja)**.
  6. Salin **Web App URL** yang dihasilkan, lalu tempel pada variabel GAS_WEB_APP_URL di dalam kode HTML file ini (baris script JavaScript).
// ===================================================
// KODE GOOGLE APPS SCRIPT (Code.gs)
// Salin seluruh kode ini ke Apps Script Google Sheets
// ===================================================

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // Buat header jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp", 
        "Nama Siswa", 
        "Kelas", 
        "Jawaban Benar", 
        "Jawaban Salah", 
        "Skor Akhir", 
        "Jumlah Pelanggaran", 
        "Nomor Soal Salah"
      ]);
    }
    
    var data = JSON.parse(e.postData.contents);
    
    sheet.appendRow([
      data.timestamp || new Date().toLocaleString("id-ID"),
      data.nama,
      data.kelas,
      data.benar,
      data.salah,
      data.skor,
      data.pelanggaran,
      data.soalSalah
    ]);
    
    return ContentService
      .createTextOutput(JSON.stringify({"result": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({"result": "error", "error": err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
